 83    def _registry_yaml_path(self) -> Path:
 84        repo_root = Path(__file__).resolve().parents[3]
 85        vendor_path = repo_root / "vendor" / "tjbot-config" / "model-registry.yaml"
 86        if vendor_path.exists():
 87            return vendor_path
 88        return Path(__file__).resolve().parent.parent / "config" / "model-registry.yaml"

 90    def _load_metadata(self) -> None:
 91        if self._metadata_loaded:
 92            return
 93
 94        yaml_path = self._registry_yaml_path()
 95        if not yaml_path.exists():
 96            raise TJBotError(f"Model registry metadata file not found: {yaml_path}")
 97
 98        with open(yaml_path, "r", encoding="utf-8") as file:
 99            data = yaml.safe_load(file) or {}
100
101        models = data.get("models", [])
102        for model in models:
103            key = model.get("key")
104            if not key:
105                raise TJBotError("Model entry missing key")
106
107            metadata = ModelMetadata(
108                type=model.get("type", ""),
109                key=key,
110                label=model.get("label", key),
111                url=model.get("url", ""),
112                folder=model.get("folder", key),
113                required=list(model.get("required", [])),
114                kind=model.get("kind"),
115                labelUrl=model.get("labelUrl"),
116                inputShape=model.get("inputShape"),
117            )
118            self.register_model(metadata)
119
120        self._metadata_loaded = True
121
122    def register_model(self, model: ModelMetadata) -> None:
123        self.registered_models[model.key] = model

125    def lookup_model(self, model_key: str) -> ModelMetadata:
126        model = self.registered_models.get(model_key)
127        if not model:
128            raise TJBotError(f'Model with key "{model_key}" not found in registry')
129        return model
130
131    def lookup_models(
132        self, model_type: Optional[ModelType] = None, installed_only: bool = False
133    ) -> List[ModelMetadata]:
134        candidates = list(self.registered_models.values())
135        if model_type is not None:
136            candidates = [m for m in candidates if m.type == model_type]
137
138        if installed_only:
139            candidates = [m for m in candidates if self.is_model_downloaded(m.key)]
140
141        return candidates

143    def get_model_cache_dir(self) -> Path:
144        return Path.home() / ".tjbot" / "models"
145
146    def get_model_cache_dir_for_type(self, model_type: str) -> Path:
147        cache_subdir = "vision" if model_type.startswith("vision.") else model_type
148        return self.get_model_cache_dir() / cache_subdir

150    def is_model_downloaded(self, model_key: str) -> bool:
151        model = self.lookup_model(model_key)
152        model_path = self.get_model_cache_dir_for_type(model.type) / model.folder
153        if not model_path.exists():
154            return False
155        return all(
156            (model_path / required_file).exists() for required_file in model.required

159    def load_model(self, model_key: str) -> ModelMetadata:
160        model = self.lookup_model(model_key)
161        if not self.is_model_downloaded(model_key):
162            self.download_model(model_key)
163        return model

165    def _download_file(self, url: str, destination: Path) -> None:
166        destination.parent.mkdir(parents=True, exist_ok=True)
167
168        parsed = urlparse(url)
169        if parsed.scheme == "file":
170            source = Path(parsed.path)
171            if not source.exists():
172                raise TJBotError(f"File URL source not found: {url}")
173            shutil.copy2(source, destination)
174            return
175
176        response = requests.get(url, timeout=60, stream=True)
177        response.raise_for_status()
178
179        with open(destination, "wb") as file:
180            for chunk in response.iter_content(chunk_size=1024 * 1024):
181                if chunk:
182                    file.write(chunk)
183
184    def _extract_tar_bz2(self, archive_path: Path, destination_dir: Path) -> None:
185        destination_dir.mkdir(parents=True, exist_ok=True)
186        with tarfile.open(archive_path, mode="r:bz2") as archive:
187            archive.extractall(path=destination_dir)
188
189    def download_model(self, model_key: str) -> None:
190        model = self.lookup_model(model_key)
191        cache_dir = self.get_model_cache_dir_for_type(model.type)
192        model_path = cache_dir / model.folder
193
194        if self.is_model_downloaded(model_key):
195            return
196
197        cache_dir.mkdir(parents=True, exist_ok=True)
198
199        temp_archive = Path("/tmp") / f"{model.key}-download"
200        self._download_file(model.url, temp_archive)
201
202        if model.url.endswith(".tar.bz2"):
203            self._extract_tar_bz2(temp_archive, cache_dir)
204            if temp_archive.exists():
205                temp_archive.unlink()
206        else:
207            model_path.mkdir(parents=True, exist_ok=True)
208            file_name = os.path.basename(urlparse(model.url).path)
209            if not file_name:
210                raise TJBotError(
211                    f"Cannot determine file name for model URL: {model.url}"
212                )
213            target_path = model_path / file_name
214            shutil.copy2(temp_archive, target_path)
215            if temp_archive.exists():
216                temp_archive.unlink()
217
218        if model.type.startswith("vision.") and model.labelUrl:
219            label_file_name = os.path.basename(urlparse(model.labelUrl).path)
220            if label_file_name:
221                label_target_path = model_path / label_file_name
222                if not label_target_path.exists():
223                    self._download_file(model.labelUrl, label_target_path)
224
225        if not self.is_model_downloaded(model_key):
226            raise TJBotError(
227                f'Model "{model_key}" download incomplete: required files missing'
228            )
