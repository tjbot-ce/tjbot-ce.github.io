216    def _validate_against_schema(self, config: Dict[str, Any]) -> None:
217        if not self.schema_config_path.exists():
218            return
219
220        try:
221            with open(self.schema_config_path, "r", encoding="utf-8") as schema_file:
222                schema = yaml.safe_load(schema_file)
223        except Exception as e:
224            raise TJBotError(
225                f"unable to read TJBot config schema from {self.schema_config_path}: {e}"
226            )
227
228        try:
229            validator_cls = validators.validator_for(schema)
230            validator_cls.check_schema(schema)
231            validator = validator_cls(schema)
232            errors = sorted(
233                validator.iter_errors(config), key=lambda err: list(err.path)
234            )
235            if errors:
236                raise ValidationError(errors[0].message)
237        except ValidationError as e:
238            raise TJBotError("invalid TJBot configuration", cause=e)
239
240    def _load_toml_file(self, path: Path) -> Dict[str, Any]:
241        try:
242            with open(path, "rb") as f:
243                data = tomllib.load(f)
244            return self._clean_config(data)
245        except Exception as e:
246            raise TJBotError(f"unable to read TOML from {path}: {e}")
247
248    def _deep_merge(self, *sources: Dict[str, Any]) -> Dict[str, Any]:
249        """Deep merge multiple configuration objects.
250        Later objects override earlier ones, but only at the leaf level."""
251        result: Dict[str, Any] = {}
252        for source in sources:
253            for key, value in source.items():
254                result_value = result.get(key)
255                # If the value is an array, replace it entirely (no merging)
256                if isinstance(value, list):
257                    result[key] = value
258                # If both values are plain objects, merge them recursively
259                elif self._is_plain_object(value) and self._is_plain_object(
260                    result_value
261                ):
262                    result[key] = self._deep_merge(result_value, value)
263                # Otherwise, replace the value
264                else:
265                    result[key] = value
266        return result
267
268    def _is_plain_object(self, value: Any) -> TypeGuard[Dict[str, Any]]:
269        """Check if a value is a plain dict (not None, not a list, etc.)"""
270        return isinstance(value, dict)
271
272    def _clean_config(self, obj: Any) -> Any:
273        """Clean configuration object to remove non-string keys."""
274        if obj is None:
275            return obj
276        if isinstance(obj, list):
277            return [self._clean_config(item) for item in obj]
278        if isinstance(obj, dict):
279            return {
280                k: self._clean_config(v) for k, v in obj.items() if isinstance(k, str)
281            }
282        return obj

284    def _validate_vision_local_models(self, local_config: Dict[str, Any]) -> None:
285        """Validate vision local backend models are properly configured."""
286        models = [
287            {"field": "object_detection_model", "expected_kind": "detection"},
288            {"field": "image_classification_model", "expected_kind": "classification"},
289            {"field": "face_detection_model", "expected_kind": "face-detection"},
290        ]
291        for model in models:
292            if not local_config.get(model["field"]):
293                raise TJBotError(
294                    f"Vision local backend: {model['field']} is required but not configured"
295                )
296
297        confidence_fields = [
298            "object_detection_confidence",
299            "image_classification_confidence",
300            "face_detection_confidence",
301        ]
302        for field in confidence_fields:
303            value = local_config.get(field)
304            if value is not None and (
305                not isinstance(value, (int, float)) or value < 0 or value > 1
306            ):
307                raise TJBotError(
308                    f"Vision local backend: {field} must be a number between 0.0 and 1.0"
309                )
310
311    def _validate_vision_backend_config(self) -> None:
312        """Validate vision backend configuration for all backend types."""
313        see = self._config.see
314        if see is None:
315            return
316
317        backend = see.backend
318        if not backend:
319            return
320
321        if backend.type == "local" and backend.local is not None:
322            self._validate_vision_local_models(backend.local.model_dump())
323            return
324
325        if (
326            backend.type == "google-cloud-vision"
327            and backend.google_cloud_vision is not None
328        ):
329            self._validate_vision_thresholds(
330                "google-cloud-vision",
331                backend.google_cloud_vision.model_dump(),
332                [
333                    "object_detection_confidence",
334                    "image_classification_confidence",
335                    "face_detection_confidence",
336                ],
337            )
338            return
339
340        if backend.type == "azure-vision" and backend.azure_vision is not None:
341            self._validate_vision_thresholds(
342                "azure-vision",
343                backend.azure_vision.model_dump(),
344                ["object_detection_confidence", "image_classification_confidence"],
345            )
346
347    def _validate_vision_thresholds(
348        self, backend_name: str, config: Dict[str, Any], fields: List[str]
349    ) -> None:
350        """Validate confidence thresholds in a backend config object."""
351        for field in fields:
352            value = config.get(field)
353            if value is not None and (
354                not isinstance(value, (int, float)) or value < 0 or value > 1
355            ):
356                raise TJBotError(
357                    f"Vision {backend_name} backend: {field} must be a number between 0.0 and 1.0"
358                )
