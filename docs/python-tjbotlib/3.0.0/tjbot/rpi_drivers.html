180class RPiBaseHardwareDriver(RPiHardwareDriver):
181    """
182    Base implementation of RPi Hardware Driver.
183    """
184
185    def __init__(self):
186        self.initialized_hardware: Set[str] = set()
187
188        # Controllers
189        self.camera_controller: Optional[CameraController] = None
190        self.microphone_controller: Optional[MicrophoneController] = None
191        self.speaker_controller: Optional[SpeakerController] = None
192        self.stt_controller: Optional[STTController] = None
193        self.tts_controller: Optional[TTSController] = None
194        self.vision_controller: Optional[VisionController] = None
195
196        # Config cache
197        self.listen_config: Optional[ListenConfig] = None
198        self.see_config: Optional[SeeConfig] = None
199        self.shine_config: Optional[ShineConfig] = None
200        self.speak_config: Optional[SpeakConfig] = None
201
202    def has_hardware(self, hardware: str) -> bool:
203        return hardware in self.initialized_hardware
204
205    def get_hardware(self) -> Set[str]:
206        return set(self.initialized_hardware)
207
208    def has_capability(self, capability: str) -> bool:
209        if capability == Capability.LISTEN:
210            return self.has_hardware(Hardware.MICROPHONE)
211        elif capability == Capability.SEE:
212            return self.has_hardware(Hardware.CAMERA)
213        elif capability == Capability.SHINE:
214            return self.has_hardware(Hardware.LED)
215        elif capability == Capability.SPEAK:
216            return self.has_hardware(Hardware.SPEAKER)
217        elif capability == Capability.WAVE:
218            return self.has_hardware(Hardware.SERVO)
219        return False
220
221    def setup_camera(self, config: SeeConfig) -> None:
222        self.camera_controller = CameraController()
223        self.see_config = config
224        width = config.camera_resolution[0] if config.camera_resolution else 1920
225        height = config.camera_resolution[1] if config.camera_resolution else 1080
226        capture_timeout = (
227            config.capture_timeout if config.capture_timeout is not None else 500
228        )
229        zero_shutter_lag = (
230            config.zero_shutter_lag if config.zero_shutter_lag is not None else False
231        )
232        self.camera_controller.initialize(
233            (width, height),
234            config.vertical_flip or False,
235            config.horizontal_flip or False,
236            capture_timeout,
237            zero_shutter_lag,
238        )
239        self.initialized_hardware.add(Hardware.CAMERA)
240
241    def setup_microphone(self, config: ListenConfig) -> None:
242        self.microphone_controller = MicrophoneController()
243        self.listen_config = config
244
245        rate = config.microphone_rate or 44100
246        channels = config.microphone_channels or 2
247        device = config.device or ""
248
249        self.microphone_controller.initialize(rate, channels, device)
250        self.initialized_hardware.add(Hardware.MICROPHONE)
251
252    def setup_led(self, config: ShineConfig) -> None:
253        self.shine_config = config
254        if config.has_common_anode_led:
255            self.setup_led_common_anode(config.common_anode or LEDCommonAnodeConfig())
256        if config.has_neopixel_led:
257            self.setup_led_neopixel(config.neopixel or LEDNeopixelConfig())
258
259    def setup_speaker(self, config: SpeakConfig) -> None:
260        self.speaker_controller = SpeakerController()
261        self.speak_config = config
262        device = config.device or ""
263
264        if not is_command_available("aplay"):
265            raise TJBotError(
266                "TJBot requires the aplay command for audio playback. "
267                "Install it with: sudo apt-get install alsa-utils"
268            )
269
270        self.speaker_controller.initialize(device)
271        self.speaker_controller.set_audio_lifecycle_callbacks(
272            lambda: self.pause_mic(), lambda: self.resume_mic()
273        )
274        self.initialized_hardware.add(Hardware.SPEAKER)
275
276    def initialize_stt_engine(self) -> None:
277        if self.microphone_controller is None:
278            raise TJBotError(
279                "Microphone controller not initialized. Call setup_microphone() before initializing STT."
280            )
281        self.stt_controller = STTController(
282            microphone_controller=self.microphone_controller
283        )
284        self.stt_controller.initialize(self.listen_config or ListenConfig())
285
286    def initialize_tts_engine(self) -> None:
287        if self.speaker_controller is None:
288            raise TJBotError(
289                "Speaker controller not initialized. Call setup_speaker() before initializing TTS."
290            )
291        self.tts_controller = TTSController(self.speaker_controller)
292        self.tts_controller.initialize(self.speak_config or SpeakConfig())
293
294    def initialize_vision_engine(self) -> None:
295        self.vision_controller = VisionController()
296        self.vision_controller.initialize(self.see_config or SeeConfig())
297
298    def capture_photo(self, file_path: Optional[str] = None) -> str:
299        if not self.camera_controller:
300            raise TJBotError("Camera not initialized.")
301        return self.camera_controller.capture_photo(file_path)
302
303    def capture_photo_buffer(self) -> bytes:
304        if not self.camera_controller:

308    def detect_objects(self, image: Any) -> Any:

362        if self.microphone_controller is None:

383    def get_mic_input_stream(self) -> Any:
384        if self.microphone_controller is None:
385            raise TJBotError(

389
390    def cleanup(self) -> None:
391        """Clean up all controllers and reset hardware state."""
392        for controller_attr in (
393            "camera_controller",
394            "microphone_controller",
395            "speaker_controller",
396            "stt_controller",
397            "tts_controller",
398            "vision_controller",
399        ):
400            controller = getattr(self, controller_attr, None)
401            if controller is not None:
402                cleanup_fn = getattr(controller, "cleanup", None)
403                if callable(cleanup_fn):
404                    try:
405                        cleanup_fn()
406                    except Exception:
407                        pass
408                setattr(self, controller_attr, None)
409
410        self.initialized_hardware.clear()
