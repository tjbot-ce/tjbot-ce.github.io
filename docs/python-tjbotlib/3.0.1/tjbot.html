190    def cleanup(self) -> None:
191        with self._cleanup_condition:
192            if self._cleanup_in_progress:
193                wait_run_id = self._cleanup_run_id
194                while self._cleanup_in_progress and self._cleanup_run_id == wait_run_id:
195                    self._cleanup_condition.wait()
196
197                if (
198                    self._cleanup_completed_run_id == wait_run_id
199                    and self._cleanup_error is not None
200                ):
201                    raise TJBotError(
202                        "Failed to clean up TJBot resources", cause=self._cleanup_error
203                    )
204                return
205
206            self._cleanup_in_progress = True
207            self._cleanup_run_id += 1
208            run_id = self._cleanup_run_id
209            self._cleanup_error = None
210
211        driver = self.rpi_driver
212        cleanup_error: Optional[Exception] = None
213        try:
214            if driver is None:
215                self._initialized = False
216                return
217
218            cleanup = getattr(driver, "cleanup", None)
219            if callable(cleanup):
220                cleanup()
221            self._initialized = False
222        except Exception as error:
223            cleanup_error = (
224                error if isinstance(error, Exception) else Exception(str(error))
225            )
226            raise TJBotError("Failed to clean up TJBot resources", cause=cleanup_error)
227        finally:
228            with self._cleanup_condition:
229                self._cleanup_in_progress = False
230                self._cleanup_completed_run_id = run_id
231                self._cleanup_error = cleanup_error
232                self._cleanup_condition.notify_all()

236    def _install_process_cleanup_hooks(self) -> None:
237        if self._process_hooks_installed:
238            return
239
240        # Python only permits signal registration in the main thread.
241        if threading.current_thread() is not threading.main_thread():
242            return
243
244        atexit.register(self._run_lifecycle_cleanup)
245
246        for sig_name in ("SIGINT", "SIGTERM", "SIGHUP"):
247            sig = getattr(signal, sig_name, None)
248            if sig is None:
249                continue
250
251            previous = signal.getsignal(sig)
252
253            def _handler(signum, frame, prev=previous, _sig_name=sig_name):
254                _ = frame
255                exit_code = {"SIGINT": 130, "SIGTERM": 143, "SIGHUP": 129}.get(
256                    _sig_name, 1
257                )
258                self._run_lifecycle_cleanup(exit_code=exit_code)
259                if callable(prev):
260                    prev(signum, frame)
261
262            signal.signal(sig, _handler)
263
264        self._process_hooks_installed = True
265
266    def _run_lifecycle_cleanup(self, exit_code: Optional[int] = None) -> None:
267        """
268        Best-effort cleanup path used by process lifecycle hooks.
269
270        When called from a signal handler (exit_code is set), cleanup runs in a
271        background thread and is given at most _CLEANUP_TIMEOUT_S seconds before
272        the process is force-exited — matching Node's Promise.race approach.
273
274        When called from atexit (exit_code is None), cleanup runs inline so the
275        process drains fully before exiting.
276        """
277        if exit_code is None:
278            # atexit: run inline, best-effort
279            try:
280                self.cleanup()
281            except Exception as error:
282                logger.warning("TJBot lifecycle cleanup failed: %s", error)
283            return
284
285        # Signal path: run cleanup in a thread with a hard timeout
286        cleanup_done = threading.Event()
287
288        def _do_cleanup():
289            try:
290                self.cleanup()
291            except Exception as error:
292                logger.warning("TJBot lifecycle cleanup failed: %s", error)
293            finally:
294                cleanup_done.set()
295
296        t = threading.Thread(target=_do_cleanup, daemon=True)
297        t.start()
298        cleanup_done.wait(timeout=self._CLEANUP_TIMEOUT_S)
299        raise SystemExit(exit_code)

301    def _initialize_hardware_from_config(self):
302        if self.rpi_driver is None:
303            return
304        config = self.config
305        if config is None:
306            return
307
308        hw_config = config.hardware
309        enabled_hardware: List[str] = []
310
311        # Determine what to init
312        if hw_config.speaker:
313            enabled_hardware.append(Hardware.SPEAKER)
314        if hw_config.microphone:
315            enabled_hardware.append(Hardware.MICROPHONE)
316        if hw_config.camera:
317            enabled_hardware.append(Hardware.CAMERA)
318        if (
319            getattr(hw_config, "led", False)
320            or getattr(hw_config, "led_neopixel", False)
321            or getattr(hw_config, "led_common_anode", False)
322        ):
323            enabled_hardware.append(Hardware.LED)
324        if hw_config.servo:
325            enabled_hardware.append(Hardware.SERVO)
326
327        if not enabled_hardware:
328            logger.debug("No hardware configured in config file")
329            return
330
331        logger.info(f"Initializing TJBot with {', '.join(enabled_hardware)}")
332
333        for hw in enabled_hardware:
334            if hw == Hardware.CAMERA:
335                self.rpi_driver.setup_camera(config.see)
336            elif hw == Hardware.LED:
337                shine_config = config.shine
338                has_neopixel = bool(
339                    (
340                        getattr(shine_config, "has_neopixel_led", False)
341                        if shine_config
342                        else False
343                    )
344                    or getattr(hw_config, "led_neopixel", False)
345                )
346                has_common_anode = bool(
347                    (
348                        getattr(shine_config, "has_common_anode_led", False)
349                        if shine_config
350                        else False
351                    )
352                    or getattr(hw_config, "led_common_anode", False)
353                )
354
355                if not has_neopixel and not has_common_anode:
356                    raise TJBotError(
357                        "LED hardware enabled but no LED type configured. Set shine.hasNeopixelLED or "
358                        "shine.hasCommonAnodeLED to true in your tjbot configuration file (~/.tjbot/tjbot.toml)."
359                    )
360
361                if has_neopixel:
362                    neopixel = shine_config.neopixel
363                    if not neopixel or (
364                        neopixel.gpio_pin is None and not neopixel.spi_interface
365                    ):
366                        raise TJBotError(
367                            "NeoPixel LED hardware is enabled but shine.neopixel is not configured. "
368                            "Define [shine.neopixel] in your tjbot configuration."
369                        )
370                    self.rpi_driver.setup_led_neopixel(neopixel)
371
372                if has_common_anode:
373                    common_anode = shine_config.common_anode
374                    if not common_anode or (
375                        common_anode.red_pin is None
376                        or common_anode.green_pin is None
377                        or common_anode.blue_pin is None
378                    ):
379                        raise TJBotError(
380                            "Common-anode LED hardware is enabled but shine.commonanode is not configured. "
381                            "Define [shine.commonanode] in your tjbot configuration."
382                        )
383                    self.rpi_driver.setup_led_common_anode(common_anode)
384            elif hw == Hardware.MICROPHONE:
385                self.rpi_driver.setup_microphone(config.listen)
386            elif hw == Hardware.SERVO:
387                self.rpi_driver.setup_servo(config.wave)
388            elif hw == Hardware.SPEAKER:
389                self.rpi_driver.setup_speaker(config.speak)

416    def shine(self, color: str) -> None:
417        """
418        Change the color of the LED.
419        :param color: Hex color, name, "on", or "off".
420        """
421        self._assert_capability(Capability.SHINE)
422        c = normalize_color(color)
423        # remove leading # for driver if needed?
424        # Drivers accept #RRGGBB or RRGGBB usually.
425        # rpi5_driver converts hex string using convert_hex_to_rgb or passes to spi.
426        # utils.normalize_color returns #RRGGBB.
427        # rpi3/rpi4 LEDNeopixel expects an int color value.
428        # Let's strip # just in case driver expects clean hex.
429        # Actually standardizing on #RRGGBB is better, but existing driver code might assume no #.
430        # rpi5_driver `render_led` -> `convert_hex_to_rgb_color` strips #. `neopixel_led.render` (SPI) parses int(color, 16) which handles 0x but maybe not #.
431        # `int("#ffffff", 16)` fails. `int("ffffff", 16)` works.
432        # So I should strip # before calling driver render_led.
433        if c.startswith("#"):
434            c = c[1:]
435
436        # Async in Node? Node `shine` is async. Python usually sync unless using asyncio.
437        # RPi driver `render_led` is sync.
438        driver = self.rpi_driver
439        if driver is None:
440            raise TJBotError(
441                "TJBot has not been initialized. Call initialize() before using TJBot methods."
442            )
443        driver.render_led(c)

445    def pulse(self, color: str, duration: float = 1.0) -> None:
446        """
447        Pulse the LED a single time.
448        """
449        self._assert_capability(Capability.SHINE)
450
451        if duration < 0.5:
452            logger.warning(
453                "TJBot cannot pulse for less than 0.5 seconds, using duration of 0.5 seconds"
454            )
455            duration = 0.5
456        if duration > 2.0:
457            logger.warning(
458                "TJBot cannot pulse for more than 2 seconds, using duration of 2.0 seconds"
459            )
460            duration = 2.0
461
462        num_steps = 20
463        # fps = num_steps / duration
464        # delay = 1.0 / fps # delay between steps
465
466        # Ease function: quadratic in-out
467        # t: current time, b: start, c: change, d: duration
468        def ease_in_out_quad(t, b, c, d):
469            t /= d / 2
470            if t < 1:
471                return c / 2 * t * t + b
472            t -= 1
473            return -c / 2 * (t * (t - 2) - 1) + b
474
475        # Generate brightness/color ramp
476        # Node impl generates a color ramp by varying lightness in HSL.
477        # We can implement similar or just fade from OFF to COLOR to OFF?
478        # Node: "colorRamp[i] = hex.toHsl().lightness(l).toRgb()..."
479        # It ramps lightness from 0.0 to 0.5.
480
481        import colorsys
482        import webcolors
483
484        rgb_target = webcolors.hex_to_rgb(normalize_color(color))  # (r, g, b)
485        # Convert to HLS (Hue, Lightness, Saturation)
486        h, lightness, s = colorsys.rgb_to_hls(
487            rgb_target.red / 255.0, rgb_target.green / 255.0, rgb_target.blue / 255.0
488        )
489
490        # We want to ramp L from 0 to 0.5 (or target L?)
491        # Node code: `l = 0.0 + (i / (numSteps / 2)) * 0.5;`
492        # This implies it peaks at L=0.5. If the color is lighter than 0.5, it might look weird?
493        # Assuming standard LED behavior, max brightness is usually decent.
494
495        ramp_colors = []
496        half_steps = int(num_steps / 2)
497        for i in range(half_steps):
498            l_val = (i / half_steps) * 0.5
499            r, g, b = colorsys.hls_to_rgb(h, l_val, s)
500            ramp_colors.append(
501                webcolors.rgb_to_hex((int(r * 255), int(g * 255), int(b * 255)))
502            )
503
504        # Full ramp: up + down
505        full_ramp = ramp_colors + ramp_colors[::-1]
506
507        # Easing logic to create delays
508        # Node creates 'ease' array of times, then diffs them to get delays.
509        ease_times = []
510        for i in range(len(full_ramp)):
511            t = ease_in_out_quad(i, 0, 1, len(full_ramp))
512            ease_times.append(t * duration)
513
514        # Delays
515        # Wait, if we use time.sleep, we need delta.
516        # Node: ease = ease.map(x => x * duration); easeDelays[i] = ease[i+1] - ease[i];
517
518        prev_time = 0
519        for i, c in enumerate(full_ramp):
520            # Calculate sleep time
521            target_time = ease_times[i]
522            sleep_time = target_time - prev_time
523            if sleep_time > 0:
524                tjbot_sleep(sleep_time)
525
526            # Render
527            if c.startswith("#"):
528                c = c[1:]
529            self.rpi_driver.render_led(c)  # type: ignore[union-attr]
530            prev_time = target_time

588    def listen(self, timeout: Optional[float] = None) -> str:
589        """
590        Listen for speech.
591
592        Args:
593            timeout: Optional timeout in seconds. If speech is not detected within this
594                time, a TJBotError is raised.
595
596        Returns:
597            The final transcript from the microphone.
598
599        Raises:
600            TJBotError: If TJBot is not initialized or the configured STT mode is streaming.
601        """
602        driver = self._assert_capability(Capability.LISTEN)
603
604        config = self.config
605        if config is None:
606            raise TJBotError(
607                "TJBot has not been initialized. Call initialize() before using TJBot methods."
608            )
609
610        listen_config = config.listen
611        mode = infer_stt_mode(listen_config)
612
613        local_cfg = (
614            listen_config.backend.local
615            if (listen_config.backend and listen_config.backend.local)
616            else None
617        )
618        if mode == "streaming":
619            model_name = (local_cfg.model if local_cfg else None) or "<unknown>"
620            raise TJBotError(
621                f'STT model "{model_name}" is streaming. Call listen_async(on_partial_result, on_final_result) '
622                f"to receive partial/final transcript callbacks."
623            )
624
625        if timeout is not None:
626            abort_event = threading.Event()
627            timer = threading.Timer(timeout, abort_event.set)
628            timer.start()
629            try:
630                result = driver.listen_for_transcript(abort_signal=abort_event)
631            finally:
632                timer.cancel()
633        else:
634            result = driver.listen_for_transcript()
635
636        logger.info(f'Heard: "{result}"')
637        return result

639    async def listen_async(
640        self,
641        on_partial_result: Optional[Callable[[str], None]] = None,
642        on_final_result: Optional[Callable[[str], None]] = None,
643        timeout: Optional[float] = None,
644    ) -> None:
645        """
646        Listen for speech asynchronously with optional callbacks for partial/final results.
647
648        Args:
649            on_partial_result: Callback invoked with each partial transcript.
650            on_final_result: Callback invoked with the final transcript.
651            timeout: Optional timeout in seconds. If speech is not detected within this
652                time, a TJBotError is raised.
653        """
654        driver = self._assert_capability(Capability.LISTEN)
655
656        config = self.config
657        if config is None:
658            raise TJBotError(
659                "TJBot has not been initialized. Call initialize() before using TJBot methods."
660            )
661
662        listen_config = config.listen
663        mode = infer_stt_mode(listen_config)
664
665        if on_partial_result is None and on_final_result is None:
666            raise TJBotError(
667                "listen_async() requires at least one callback. Use listen() for synchronous final transcript mode."
668            )
669
670        abort_event = threading.Event() if timeout is not None else None
671        timer = (
672            threading.Timer(timeout, abort_event.set)
673            if timeout is not None and abort_event is not None
674            else None
675        )
676        if timer is not None:
677            timer.start()
678
679        try:
680            if mode == "streaming":
681                loop = asyncio.get_running_loop()
682
683                def _dispatch_callback(
684                    callback: Optional[Callable[[str], None]], text: str
685                ) -> None:
686                    if callback is None:
687                        return
