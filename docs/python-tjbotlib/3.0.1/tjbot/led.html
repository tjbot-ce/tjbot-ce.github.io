20class LEDCommonAnode:
21    """
22    LED controller for Common Anode LEDs using gpiozero.
23    """
24
25    def __init__(self, red_pin: int, green_pin: int, blue_pin: int):
26        # gpiozero handles common anode (active_high=False)
27        # However, Node implementation says "0 is ON, 255 is OFF" which implies active_low logic (sink).
28        # gpiozero's RGBLED defaults to active_high=True (source).
29        # If common anode, we connect anode to 3.3V, and pins to cathodes.
30        # Pulling pin LOW turns LED ON.
31        # So active_high should be False.
32        self.led = RGBLED(
33            red=red_pin, green=green_pin, blue=blue_pin, active_high=False
34        )
35
36    def render(self, rgb_color: Tuple[int, int, int]) -> None:
37        """
38        Render the LED to a specific RGB color.
39        :param rgb_color: RGB color as (red, green, blue) where each is 0-255.
40        """
41        # gpiozero expects 0-1 floats
42        r = rgb_color[0] / 255.0
43        g = rgb_color[1] / 255.0
44        b = rgb_color[2] / 255.0
45        self.led.color = (r, g, b)
46
47    def cleanup(self) -> None:
48        self.led.close()

130    def _do_init(self, pin: int) -> None:
131        try:
132            self._send_and_wait({"cmd": "init", "pin": pin, "numLeds": 1}, timeout=10.0)
133            self._ready.set()
134        except Exception as e:
135            self._ready_error = e
136            self._ready.set()
137
138    def _send_and_wait(self, payload: dict, timeout: float) -> None:
139        """Write a command to the helper and block until the response arrives."""
140        if self._helper_dead is not None:
141            raise self._helper_dead
142
143        future: Future = Future()
144        with self._pending_lock:
145            msg_id = self._next_id
146            self._next_id += 1
147            self._pending[msg_id] = future
148
149        try:
150            data = json.dumps({**payload, "id": msg_id}) + "\n"
151            if self._helper is None or self._helper.stdin is None:
152                raise TJBotError(
153                    "NeoPixel helper process is not running or stdin is not available"
154                )
155            self._helper.stdin.write(data.encode())
156            self._helper.stdin.flush()
157        except Exception as exc:
158            with self._pending_lock:
159                self._pending.pop(msg_id, None)
160            raise TJBotError(
161                f"failed to send '{payload.get('cmd')}' to NeoPixel helper: {exc}"
162            ) from exc
163
164        try:
165            future.result(timeout=timeout)
166        except TimeoutError:
167            with self._pending_lock:
168                self._pending.pop(msg_id, None)
169            raise TJBotError(
170                f"NeoPixel helper timed out waiting for '{payload.get('cmd')}' "
171                f"response ({timeout}s)"
172            )
173
174    def _read_loop(self) -> None:
175        if self._helper is None or self._helper.stdout is None:
176            raise TJBotError(
177                "NeoPixel helper process is not running or stdout is not available"
178            )
179        try:
180            for raw in self._helper.stdout:
181                line = raw.decode("utf-8", errors="replace").strip()
182                if line:
183                    self._handle_response(line)
184        except Exception:
185            pass
186
187        # Helper process exited.
188        code = self._helper.wait()
189        stderr_summary = " | ".join(self._stderr_tail)
190        msg = f"NeoPixel helper exited unexpectedly (code={code})"
191        if stderr_summary:
192            msg += f"; stderr: {stderr_summary}"
193        self._helper_dead = TJBotError(msg)
194
195        with self._pending_lock:
196            pending = dict(self._pending)
197            self._pending.clear()
198        for fut in pending.values():
199            if not fut.done():
200                fut.set_exception(self._helper_dead)
201
202    def _stderr_loop(self) -> None:
203        if self._helper is None or self._helper.stderr is None:
204            raise TJBotError(
205                "NeoPixel helper process is not running or stderr is not available"
206            )
207        try:
208            for raw in self._helper.stderr:
209                line = raw.decode("utf-8", errors="replace").rstrip()
210                if line:
211                    sys.stderr.write(line + "\n")
212                    self._stderr_tail.append(line)
213                    if len(self._stderr_tail) > 8:
214                        self._stderr_tail = self._stderr_tail[-8:]
215        except Exception:
216            pass
217
218    def _handle_response(self, line: str) -> None:
219        try:
220            msg = json.loads(line)
221        except json.JSONDecodeError:
222            _logger.warning("NeoPixel helper sent unparseable response: %s", line)
223            return
224
225        msg_id = msg.get("id")
226        with self._pending_lock:
227            future = self._pending.pop(msg_id, None)
228
229        if future is None or future.done():
230            return
231
232        if msg.get("ok"):
233            future.set_result(None)
234        else:
235            future.set_exception(
236                TJBotError(f"NeoPixel helper error: {msg.get('error', 'unknown')}")
237            )
238
239    def _kill_helper(self) -> None:
240        if self._helper is not None and self._helper.poll() is None:
241            try:
242                if self._helper.stdin:
243                    self._helper.stdin.close()
244            except Exception:
245                pass
246            try:
247                self._helper.terminate()
248            except Exception:
249                pass

class LEDNeopixelSPI: 
 25class LEDNeopixelSPI:
 26    """
 27    LED controller for SPI-based NeoPixel LEDs (Raspberry Pi 5).
 28    This is based on pi5neo.py:
 29    https://github.com/vanshksingh/Pi5Neo/blob/main/pi5neo/pi5neo.py
 30    """
 31
 32    HIGH = 0xF8  # possibles: F0, F8, FC
 33    LOW = 0xC0  # possibles: C0
 34    FREQ = 6400000  # possibles: 3200000, 6400000; pi5neo uses: spi_speed_khz (800) * 1024 * 8 = 6553600
 35
 36    def __init__(
 37        self, spi_interface: str = "/dev/spidev0.0", use_grb_format: bool = False
 38    ):
 39        if spidev is None:
 40            raise ImportError("spidev library not found. Please install it.")
 41
 42        self.use_grb_format = use_grb_format
 43
 44        # Parse spi interface string "/dev/spidevX.Y"
 45        try:
 46            parts = spi_interface.split("spidev")[1].split(".")
 47            bus = int(parts[0])
 48            device = int(parts[1])
 49        except (IndexError, ValueError):
 50            # Fallback defaults
 51            bus = 0
 52            device = 0
 53
 54        self.spi = spidev.SpiDev()
 55        self.spi.open(bus, device)
 56        self.spi.max_speed_hz = self.FREQ
 57
 58    @staticmethod
 59    def _byte_to_bitstream(byte: int) -> List[int]:
 60        bitstream = []
 61        for i in range(8):
 62            # MSB first
 63            if (byte & (1 << (7 - i))) != 0:
 64                bitstream.append(LEDNeopixelSPI.HIGH)
 65            else:
 66                bitstream.append(LEDNeopixelSPI.LOW)
 67        return bitstream
 68
 69    @staticmethod
 70    def _rgb_to_spi_bitstream(
 71        red: int, green: int, blue: int, use_grb: bool
 72    ) -> List[int]:
 73        red_bits = LEDNeopixelSPI._byte_to_bitstream(red)
 74        green_bits = LEDNeopixelSPI._byte_to_bitstream(green)
 75        blue_bits = LEDNeopixelSPI._byte_to_bitstream(blue)
 76
 77        if use_grb:
 78            return green_bits + red_bits + blue_bits
 79        else:
 80            return red_bits + green_bits + blue_bits
 81
 82    def render(self, color: str) -> None:
 83        """
 84        Render the LED to a specified color.
 85        :param color: Hex color string in RRGGBB format with no leading '0x' or '#'.
 86
 87        Note: This method blocks until the SPI transfer completes to ensure
 88        timing integrity of the WS2812B protocol on RPi 5.
 89        """
 90        c = int(color, 16)
 91        r = (c & 0xFF0000) >> 16
 92        g = (c & 0x00FF00) >> 8
 93        b = (c & 0x0000FF) >> 0
 94
 95        bitstream = self._rgb_to_spi_bitstream(r, g, b, self.use_grb_format)
 96
 97        # Transfer data via SPI to update the LED.
 98        # The writebytes() call blocks until the transfer completes.
 99        self.spi.writebytes(bitstream)
100
101        # Add delay to ensure the SPI transfer is fully complete and the LED
102        # has latched the data. WS2812B requires ~50µs reset time minimum,
103        # but we use 10ms to account for SPI overhead and ensure stability.
104        time.sleep(0.01)  # 10ms delay for reliable data latching
105
106    def cleanup(self) -> None:
107        if self.spi:
108            self.spi.close()
